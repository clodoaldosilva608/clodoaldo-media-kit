    91→def emotional_hooks_wheel() -> str:
    92→    """Roda de ganchos emocionais."""
    93→    fig, ax = plt.subplots(figsize=(8, 8))
    94→    
    95→    emotions = [
    96→        ("Curiosidade", "#C8A062"),
    97→        ("Surpresa", "#D9B885"),
    98→        ("Medo/Urgência", "#C97064"),
    99→        ("Raiva", "#9B7A45"),
   100→        ("Tristeza", "#7B6A5A"),
   101→        ("Alegria", "#D4A574"),
   102→        ("Aspiração", "#C8A062"),
   103→        ("Pertencimento", "#B8945A"),
   104→    ]
   105→    
   106→    n = len(emotions)
   107→    angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
   108→    
   109→    for i, (label, color) in enumerate(emotions):
   110→        angle = angles[i]
   111→        wedge = Wedge((0, 0), 1, np.degrees(angle), np.degrees(angles[(i + 1) % n]),
   112→                      facecolor=color, edgecolor="#1F1A14", linewidth=2, alpha=0.85)
   113→        ax.add_patch(wedge)
   114→        
   115→        mid_angle = (angle + angles[(i + 1) % n]) / 2
   116→        x = 0.7 * np.cos(mid_angle)
   117→        y = 0.7 * np.sin(mid_angle)
   118→        ax.text(x, y, label, ha="center", va="center",
   119→                fontsize=11, fontweight="bold", color="#1F1A14")
   120→    
   121→    # Círculo central
   122→    central = Circle((0, 0), 0.25, facecolor="#1F1A14", edgecolor="#C8A062", linewidth=2)
   123→    ax.add_patch(central)
   124→    ax.text(0, 0, "Ganchos\nEmocionais", ha="center", va="center",
   125→            fontsize=11, fontweight="bold", color="#C8A062")
   126→    
   127→    ax.set_xlim(-1.3, 1.3)
   128→    ax.set_ylim(-1.3, 1.3)
   129→    ax.set_aspect("equal")
   130→    ax.axis("off")
   131→    ax.set_title("8 Ganchos Emocionais Universais", pad=20, color="#D9B885", fontsize=13)
   132→    plt.tight_layout()
   133→    return save_illustration(fig, "emotional_hooks_wheel")

   884→def color_grading_wheel() -> str:
   885→    """Roda de color grading — looks populares."""
   886→    fig, ax = plt.subplots(figsize=(9, 8))
   887→    
   888→    looks = [
   889→        ("Teal & Orange", "#1F5A6E", "#D4762A", "Cinema clássico\nSombras frias / pele quente"),
   890→        ("Bronze Warm", "#9B7A45", "#F5EFE0", "Tom premium\nMarca pessoal"),
   891→        ("Noir B&W", "#2A231C", "#F5EFE0", "Dramático\nAlto contraste"),
   892→        ("Vintage Film", "#7B5A3E", "#E8C99B", "Nostálgico\nGrão + fade"),
   893→        ("Cyber Neon", "#3A1F4E", "#D9A574", "Futurista\nHightlights saturados"),
   894→        ("Documentário", "#5A4E3A", "#E8DCC4", "Natural\nMinimalista"),
   895→    ]
   896→    
   897→    n = len(looks)
   898→    angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
   899→    
   900→    for i, (name, c1, c2, desc) in enumerate(looks):
   901→        angle = angles[i]
   902→        mid = angle + (2 * np.pi / n) / 2
   903→        
   904→        # Wedge com gradiente simulado
   905→        wedge = Wedge((0, 0), 1.2, np.degrees(angle), np.degrees(angles[(i + 1) % n]),
   906→                      facecolor=c1, edgecolor="#1F1A14", linewidth=2, alpha=0.85)
   907→        ax.add_patch(wedge)
   908→        
   909→        x = 0.85 * np.cos(mid)
   910→        y = 0.85 * np.sin(mid)
   911→        ax.text(x, y, name, ha="center", va="center",
   912→                fontsize=10, fontweight="bold", color=c2)
   913→        
   914→        x2 = 1.5 * np.cos(mid)
   915→        y2 = 1.5 * np.sin(mid)
   916→        ax.text(x2, y2, desc, ha="center", va="center",
   917→                fontsize=8, color="#F5EFE0", style="italic")
   918→    
   919→    # Centro
   920→    central = Circle((0, 0), 0.35, facecolor="#1F1A14", edgecolor="#C8A062", linewidth=2)
   921→    ax.add_patch(central)
   922→    ax.text(0, 0, "Color\nGrading", ha="center", va="center",
   923→            fontsize=10, fontweight="bold", color="#C8A062")
   924→    
   925→    ax.set_xlim(-2.2, 2.2)
   926→    ax.set_ylim(-2.2, 2.2)
   927→    ax.set_aspect("equal")
   928→    ax.axis("off")
   929→    ax.set_title("6 Looks de Color Grading Populares",
   930→                 pad=15, color="#D9B885", fontsize=13, fontweight="bold")
   931→    plt.tight_layout()
   932→    return save_illustration(fig, "color_grading_wheel")

   935→def retention_dropoff_chart() -> str:
   936→    """Gráfico: onde a audiência abandona o vídeo."""
   937→    fig, ax = plt.subplots(figsize=(10, 5))
   938→    
   939→    t = np.linspace(0, 60, 100)
   940→    # Retenção típica de Reels de 60s
   941→    base = 100 * np.exp(-t / 35)
   942→    # Adiciona o "death valley" entre 15-25s
   943→    valley = -15 * np.exp(-((t - 20) ** 2) / 30)
   944→    # Boost no final com CTA
   945→    boost = 8 * np.exp(-((t - 55) ** 2) / 30)
   946→    retention = base + valley + boost
   947→    retention = np.clip(retention, 0, 100)
   948→    
   949→    ax.fill_between(t, 0, retention, color="#C8A062", alpha=0.2)
   950→    ax.plot(t, retention, color="#D9B885", linewidth=3)
   951→    
   952→    # Anotações das zonas críticas
   953→    ax.annotate("Gancho\n0-3s", xy=(1.5, retention[2]), xytext=(3, 95),
   954→                fontsize=9, color="#7BB081", fontweight="bold",
   955→                arrowprops=dict(arrowstyle="->", color="#7BB081"))
   956→    ax.annotate("Death valley\n15-25s", xy=(20, retention[33]), xytext=(28, 75),
   957→                fontsize=9, color="#C97064", fontweight="bold",
   958→                arrowprops=dict(arrowstyle="->", color="#C97064"))
   959→    ax.annotate("Boost CTA\n55-60s", xy=(56, retention[93]), xytext=(38, 55),
   960→                fontsize=9, color="#D9B885", fontweight="bold",
   961→                arrowprops=dict(arrowstyle="->", color="#D9B885"))
   962→    
   963→    ax.set_xlim(0, 60)
   964→    ax.set_ylim(0, 105)

   966→    ax.set_ylabel("% da audiência", fontsize=10, color="#A39684")
